## XDC.DS are a set of timing exceptions that are necessary/required *only*
## if the eMMC DS pin is implemented, but not otherwise.  They are based upon
## a nominal system clock rate with a period of 10ns
@STRB=u_@$(PREFIX)_frontend/GEN_DATASTROBE
@BOOTACT=u_@$(PREFIX)/u_control/BOOT_LOGIC.r_boot_active
@XDC.DS=
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/mem_reg*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/GEN_REGISTERED_READ.o_rd_data*}] 9
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/mem_reg*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/GEN_REGISTERED_READ.o_rd_data*}] 9
#
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/wgray_r*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/wgray_cross*}] 10
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/wgray_r*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/wgray_cross*}] 10
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/rgray*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_nedge_fifo*/rgray_cross*}] 10
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/rgray*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/rgray_cross*}] 10
set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/rgray*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pedge_fifo*/rgray_cross*}] 10
##
## set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(BOOTACT)*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pcmd_fifo_?/wgray*}] 10
## set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(BOOTACT)*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pcmd_fifo_?/wgray_cross*}] 10
## set_max_delay -datapath_only -from [get_cells -hier -filter {NAME=~@$(BOOTACT)*}] -to [get_cells -hier -filter {NAME=~@$(STRB).u_pcmd_fifo_?/wr_rgray*}] 10
##

@XDC.INSERT=

set_property -dict { PULLTYPE PULLUP } [get_ports io_@$(TOP.IOPREFIX)_cmd]

@$(XDC.DS)
