@PREFIX=emmc
@INCLUDEFILE=sdio.txt
@DEVID=EMMC
@NAME=eMMC Card
@$NUMIO=8
@CARD_DETECT=1'b0
@OPT_SERDES=1'b1
@OPT_DDR=1'b0
@OPT_DMA=1'b0
@OPT_EMMC=1'b1
@OPT_1P8V=1'b0
@HW1P8V=1'b1
@TOP.IOPREFIX=emmc
@ACCESS=EMMC_ACCESS
@INT.SDCARD.WIRE=
@INT.SDCARD.PIC=
@INT.EMMC.WIRE= @$(PREFIX)_int
@INT.EMMC.PIC= altpic
@SDIO.CLKDECL=
		o_@$(PREFIX)_clk,
@SDIO.RSTDECL=
		o_@$(PREFIX)_reset_n,
@SDIO.IORST=
	output	wire	o_@$(PREFIX)_reset_n;
@SDIO.IOCLK=
	output	wire	o_@$(PREFIX)_clk;
@SDIO.CDETDECL=
@SDIO.IOCDET=
@SDIO.DSDECL=
		i_@$(PREFIX)_ds,
@SDIO.DSASSIGN=
	BUFG u_@$(PREFIX)_ds (
		.I(i_@$(PREFIX)_ds),
		.O(w_@$(PREFIX)_ds)
	);

	// assign	w_@$(PREFIX)_ds = i_@$(PREFIX)_ds;
@SDIO.IODS=
	input	wire		i_@$(PREFIX)_ds;
@TOP.DEFNS=
@$(SDIO.TOP.DEFNS)
	wire	[3:0]	su_@$(PREFIX)_nc;	// Unused / noconnect wires
	wire	i_@$(TOP.IOPREFIX)_cd_n;

	assign	i_@$(TOP.IOPREFIX)_cd_n = 1'b0;
	assign	o_@$(TOP.IOPREFIX)_reset_n = w_@$(PREFIX)_hwreset_n;
@SDIO.CLKASSIGN=
	assign	o_@$(TOP.IOPREFIX)_clk = w_@$(PREFIX)_ck;
@REGS.N=5
@REGS.NOTE= // @$(NAME) addresses
@REGS.0= 0 R_@$(DEVID)_CTRL  	EMMCCARD
@REGS.1= 1 R_@$(DEVID)_DATA 	EMMCDATA
@REGS.2= 2 R_@$(DEVID)_FIFOA	EMMCFIFOA, EMMCFIF0, EMMCFIFA
@REGS.3= 3 R_@$(DEVID)_FIFOB	EMMCFIFOB, EMMCFIF1, EMMCFIFB
@REGS.4= 4 R_@$(DEVID)_PHY	EMMCPHY
@BDEF.DEFN=
////////////////////////////////////////////////////////////////////////////////
//
// @$(NAME) constants
// {{{
////////////////////////////////////////////////////////////////////////////////
//
//

struct @$(DEVID)_S;
// }}}
@BDEF.IONAME=_@$(PREFIX)
@RTL.MAKE.GROUP= EMMC
@RTL.MAKE.SUBD=sdspi
@RTL.MAKE.FILES= sdio_top.v sdio.v sdfrontend.v sdckgen.v sdwb.v sdtxframe.v sdrxframe.v
@SIM.CLOCK=clk
@SIM.FILE="emmc.img"
@XDC.INSERT=
set_property -dict { PULLTYPE PULLUP } [get_ports io_@$(TOP.IOPREFIX)_cmd]
# set_property CLOCK_DEDICATED_ROUTE FALSE [get_nets i_@$(PREFIX)_ds]
@$(XDC.SERDES)
