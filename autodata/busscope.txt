@PREFIX=busscope
@DEVID=BUSSCOPE
@TGT3=wbu
@TGT2=wbwide_zip
@TGT1=wbwide_wbdown
@TGT0=wbwide_ddr3_controller
@TARGET=@$(PREFIX)
@TRIGGER=@$(TARGET)_debug[31]
@$LOG_CAPTURE_SIZE=12
@INCLUDEFILE=wbscopc.txt
@INT.BUSSCOPE.PIC=altpic
@INT.BUSSCOPE.WIRE=@$(PREFIX)_int
@MAIN.DEFNS=
	reg	[31:0]	@$(PREFIX)_debug;
@MAIN.INSERT=
	always @(*)
	begin
		@$(PREFIX)_debug = {
			@$(TGT0)_err || wbu_err || genbus.r_wdt_reset,
			//
			6'h0,
			//
			genbus.r_wdt_reset,
			@$(TGT3)_cyc,
			@$(TGT3)_stb,
			@$(TGT3)_we,
			@$(TGT3)_stall,
			@$(TGT3)_ack,
			@$(TGT3)_err,
			//
			@$(TGT2)_cyc,
			@$(TGT2)_stb,
			@$(TGT2)_we,
			@$(TGT2)_stall,
			@$(TGT2)_ack,
			@$(TGT2)_err,
			//
			@$(TGT1)_cyc,
			@$(TGT1)_stb,
			@$(TGT1)_we,
			@$(TGT1)_stall,
			@$(TGT1)_ack,
			@$(TGT1)_err,
			//
			@$(TGT0)_cyc,
			@$(TGT0)_stb,
			@$(TGT0)_we,
			@$(TGT0)_stall,
			@$(TGT0)_ack,
			@$(TGT0)_err
		};
	end

@$(SCOPE.INSERT)
@MAIN.ALT=
	initial	@$(PREFIX)_debug = 32'h0;
